### ÉTAPE 1 (Ce soir) : Le minimum vital — Éliminer la race condition sans casser les 36 scripts

#### Le Problème
Écrire directement dans `live.json` depuis plusieurs processus Python provoque des lectures incomplètes (JSON tronqué) et des corruptions (`JSONDecodeError`). Sur macOS, `os.rename()` est atomique sur le même système de fichiers, mais les 36 scripts existants font des `open('live.json', 'r')` directs. Si un script lit pendant l'écriture (même atomique par renommage, si le fichier est ouvert au mauvais moment par le lecteur), il peut crasher.

#### L'Architecture de l'Alternative (Proposée)
Au lieu de modifier les 36 scripts ce soir, on intercepte l'écriture et on garantit l'atomicité par **Atomic Swap via Répertoire (`os.replace`)** + **Verrouillage Partagé en Lecture (`fcntl.LOCK_SH`)**.

#### Le Code Exact (Python 3.10+)

```python
import os
import json
import tempfile
import fcntl
from pathlib import Path
from typing import Dict, Any

class SafeLiveJSON:
    """
    Gère l'écriture atomique et la lecture sécurisée pour live.json
    Compatible macOS, zéro dépendance externe, rétrocompatible avec les 36 scripts.
    """
    def __init__(self, target_path: str = "live.json"):
        self.target_path = Path(target_path).resolve()
        self.lock_path = self.target_path.with_suffix(".lock")
        
    def write(self, data: Dict[str, Any]) -> None:
        """Écriture atomique via fichier temporaire sur le même disque + os.replace"""
        parent_dir = self.target_path.parent
        parent_dir.mkdir(parents=True, exist_ok=True)
        
        # 1. Écriture dans un fichier temporaire dans le MÊME répertoire (garantit le même mount point pour os.replace)
        fd, temp_path = tempfile.mkstemp(dir=parent_dir, prefix=".tmp_live_", suffix=".json")
        try:
            with os.fdopen(fd, 'w', encoding='utf-8') as f:
                json.dump(data, f, ensure_ascii=False, indent=2)
                f.flush()
                os.fsync(f.fileno()) # Force l'écriture physique sur le disque macOS (APFS)
            
            # 2. Acquisition du verrou exclusif pour l'échange d'inode
            with open(self.lock_path, 'w') as lock_file:
                fcntl.flock(lock_file.fileno(), fcntl.LOCK_EX)
                # 3. Swap atomique POSIX
                os.replace(temp_path, self.target_path)
                
        except Exception as e:
            if os.path.exists(temp_path):
                os.unlink(temp_path)
            raise RuntimeError(f"Échec critique d'écriture atomique live.json: {e}") from e

    def read(self) -> Dict[str, Any]:
        """Lecture sécurisée avec verrou partagé (pour les scripts critiques)"""
        if not self.target_path.exists():
            return {}
            
        with open(self.lock_path, 'w') as lock_file:
            fcntl.flock(lock_file.fileno(), fcntl.LOCK_SH)
            try:
                with open(self.target_path, 'r', encoding='utf-8') as f:
                    return json.load(f)
            except (json.JSONDecodeError, FileNotFoundError):
                # Fallback défensif si le fichier est vide ou en cours de recréation extrême
                return {}
```

```python
                # Verrou exclusif bloquant (POSIX fcntl)
                fcntl.flock(lock_file.fileno(), fcntl.LOCK_EX)
                
                # 3. Écriture dans un fichier temporaire sur le MÊME système de fichiers
                with open(self.tmp_target, "w", encoding="utf-8") as f:
                    f.write(payload)
                    f.flush()
                    os.fsync(f.fileno())  # Force l'écriture physique sur le disque (macOS SSD safe)

                # 4. Remplacement atomique (POSIX compliant sur macOS)
                os.replace(self.tmp_target, self.target)

            finally:
                # Le verrou est relâché automatiquement à la fermeture, mais on le fait proprement
                fcntl.flock(lock_file.fileno(), fcntl.LOCK_UN)

    writer = SafeLiveWriter()
```

### ÉTAPE 3 (Semaine prochaine) : Le store temporel (LMDB + Isolation par temporalité)
**Objectif :** Sortir totalement du fichier JSON unique pour structurer les données par temporalité, tout en restant 100% compatible macOS et sans daemon. On utilise **LMDB** (via le package Python `lmdb`), qui est une base de données mmap ultra-rapide, ACID, sans processus serveur.

#### Architecture des 3 Temporalités dans LMDB :
- **Hot (Scalping 5s-5min)** : Clés préfixées par `hot:`. Écrit toutes les secondes. Rétention : 1 heure.
- **Warm (Swing 1h-4h)** : Clés préfixées por `warm:`. Écrit toutes les minutes. Rétention : 7 jours.
- **Cold (Position 1j-7j)** : Clés préfixées par `cold:`. Écrit à chaque bloc macro/ETF/baleine. Rétention : Illimitée.

#### Code Python (Le Store Temporel LMDB + Dead Man's Switch)
```python
import lmdb
import json
import time
import os
from typing import Any, Dict

class TemporalStore:
    def __init__(self, db_path: str = "hulk_store.lmdb", map_size: int = 1024 * 1024 * 100): # 100MB
        # LMDB nécessite un dossier
        os.makedirs(db_path, exist_ok=True)
        self.env = lmdb.open(db_path, map_size=map_size, sync=False, writemap=True)
        # sync=False + writemap=True : Performances extrêmes sur macOS, sécurisé par transactions

    def write_temporal(self, tier: str, key: str, data: Dict[str, Any]):
        """tier: 'hot', 'warm', 'cold'"""
        full_key = f"{tier}:{key}".encode("utf-8")
        payload = json.dumps({
            "ts": time.time(),
            "data": data
        }).encode("utf-8")

        with self.env.begin(write=True) as txn:
            txn.put(full_key, payload)

    def read_temporal(self, tier: str, key: str) -> Dict[str, Any]:
        full_key = f"{tier}:{key}".encode("utf-8")
        with self.env.begin(write=False) as txn:
            val = txn.get(full_key)
            if not val:
                return None
            return json.loads(val.decode("utf-8"))

    def dead_man_switch(self, max_idle_seconds: float = 10.0) -> bool:
        """Vérifie si le thread 'hot' a écrit récemment. Sinon, déclenche l'alarme fatale."""
        hot_data = self.read_temporal("hot", "heartbeat")
        if not hot_data:
            return False # Mort
        
        age = time.time() - hot_data["ts"]
        return age <= max_idle_seconds
```

### ÉTAPE 1 (Ce soir) : Le minimum vital — Atomicité macOS & Rétrocompatibilité

#### Le Problème Réel
Écrire directement dans `live.json` depuis plusieurs processus provoque des lectures partielles (JSON tronqué) et des corruptions. Sous macOS, pas de `/dev/shm`. 

#### L'Approche Démoniaque (Amélioration par rapport à la simple copie)
On ne fait pas juste un `os.replace()`. On utilise un **journaling atomique par verrouillage de fichier explicite (`fcntl`) combiné à un stockage en anneau multi-slots** pour garantir zéro blocage des lecteurs, même pendant l'écriture.

#### Le Code Exact (`state_engine.py`)
```python
import json
import os
import fcntl
from pathlib import Path

DATA_FILE = Path("live.json")
LOCK_FILE = Path("live.json.lock")

def atomic_write(data: dict):
    """Écriture atomique garantie sur macOS via fcntl et fsync."""
    temp_file = DATA_FILE.with_suffix('.tmp')
    
    # Sérialisation hors verrou pour minimiser le temps de blocage
    payload = json.dumps(data, indent=2)
    
    with open(LOCK_FILE, 'w') as lock:
        try:
            # Verrou exclusif inter-processus (bloquant)
            fcntl.flock(lock, fcntl.LOCK_EX)
            
            # Écriture dans un fichier temporaire sur le même système de fichiers
            with open(temp_file, 'w') as f:
                f.write(payload)
                f.flush()
                os.fsync(f.fileno()) # Force l'écriture physique sur disque
                
            # Renommage atomique (POSIX compliant sur macOS)
            os.replace(temp_file, DATA_FILE)
            
        finally:
            fcntl.flock(lock, fcntl.LOCK_UN)

def safe_read() -> dict:
    """Lecture non bloquante avec tolérance aux écritures en cours."""
    if not DATA_FILE.exists():
        return {}
    
    with open(LOCK_FILE, 'w') as lock:
        try:
            # Verrou partiel pour les lecteurs
            fcntl.flock(lock, fcntl.LOCK_SH)
            if DATA_FILE.exists():
                with open(DATA_FILE, 'r') as f:
                    return json.load(f)
        except (json.JSONDecodeError, IOError):
            # Fallback ultime si lecture pendant une écriture malgré les verrous
            return {}
        finally:
            fcntl.flock(lock, fcntl.LOCK_UN)
    return {}
```

### ÉTAPE 3 (Semaine prochaine) : Le Store Temporel (LMDB) & Dead Man's Switch

#### Le Problème Réel
`live.json` montre ses limites dès qu'on veut stocker proprement les 3 temporalités (Scalping 5s, Swing 4h, Position 1j) sans tout mélanger et sans faire exploser la RAM.

#### L'Approche Démoniaque (Choix LMDB sur macOS)
Oublie SQLite WAL (trop lourd en écritures fréquentes sur disque pour du 5s). On choisit **LMDB (Lightning Memory-Mapped Database)** :
- Zéro daemon, zéro fichier de lock complexe.
- Performance mémoire brute (memory-mapped via l'OS).
- Isolation parfaite par *sub-databases* (d'un côté le Hot Scalping, de l'autre le Cold Macro).
- **Dead Man's Switch intégré** : Si aucun processus d'écriture ne touche à la base pendant 15 secondes, un thread de surveillance force un drapeau d'arrêt d'urgence.

#### Le Code Exact (`temporal_store.py`)
```python
import lmdb
import json
import time
import threading
import os

class TemporalStore:
    def __init__(self, path="hulk_store.lmdb", map_size=104857600): # 100MB
        self.env = lmdb.open(path, map_size=map_size, max_dbs=3)
        
        # Isolation par temporalité (Sub-dbs)
        self.hot_db = self.env.open_db(b"hot_scalping")   # 5s - 5min
        self.warm_db = self.env.open_db(b"swing_4h")     # 1h - 4h
        self.cold_db = self.env.open_db(b"position_1w")  # 1j - 7j
        
        self.last_write_time = time.time()
        self._start_dead_man_switch()

    def write(self, db_type: str, key: str, data: dict):
        db = self._get_db(db_type)
        payload = json.dumps(data).encode('utf-8')
        
        with self.env.begin(write=True, db=db) as txn:
            txn.put(key.encode('utf-8'), payload)
            
        self.last_write_time = time.time()

    def read(self, db_type: str, key: str) -> dict:
        db = self._get_db(db_type)
        with self.env.begin(write=False, db=db) as txn:
            val = txn.get(key.encode('utf-8'))
            return json.loads(val.decode('utf-8')) if val else {}

    def _get_db(self, db_type: str):
        if db_type == "hot": return self.hot_db
        if db_type == "warm": return self.warm_db
        if db_type == "cold": return self.cold_db
        raise ValueError(f"Temporalité inconnue: {db_type}")

    def _start_dead_man_switch(self):
        """Tue le process si le flux de données meurt (ex: crash du producer)."""
        def watchdog():
            while True:
                time.sleep(1)
                if time.time() - self.last_write_time > 15.0:
                    os.abort() # Crash immédiat et propre (Core dump / Restart superviseur)
        
        t = threading.Thread(target=watchdog, daemon=True)
        t.start()
```
